Scenario and rate rules
• Standard rate: Rs. 30.00 per litre at 3.5% fat and 8.5% SNF.
• Every 0.1% fat step above or below 3.5% changes the rate by Rs. 0.40.
• Every 0.1% SNF step above or below 8.5% changes the rate by Rs. 0.30.
• Farmers are paid every ten days.
The assignment specifies fat validation from 2.0% to 7.0%. SNF is treated as a positive numeric value in this solution; any
additional operational SNF limits should be confirmed by the co-operative.

TASK 9: Implement and Test the Calculator (4 Marks)
The following complete program uses Decimal from strings, validates quantity/fat/SNF and adds each rounded payment
to the ten-day total.
from decimal import Decimal, ROUND_HALF_UP
BASE_RATE = Decimal("30.00")
FAT_STANDARD = Decimal("3.5")
SNF_STANDARD = Decimal("8.5")
FAT_STEP_VALUE = Decimal("0.40")
SNF_STEP_VALUE = Decimal("0.30")
STEP = Decimal("0.1")
MIN_FAT = Decimal("2.0")
MAX_FAT = Decimal("7.0")
def money(value):
    """Round a monetary value to two decimal places."""
    return value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
def calculate_rate(fat, snf):
    """Calculate the milk rate per litre."""
    fat_steps = ((fat - FAT_STANDARD) / STEP).quantize(
        Decimal("1"), rounding=ROUND_HALF_UP)
    snf_steps = ((snf - SNF_STANDARD) / STEP).quantize(
        Decimal("1"), rounding=ROUND_HALF_UP)
    rate = BASE_RATE + fat_steps * FAT_STEP_VALUE + snf_steps * SNF_STEP_VALUE
    return money(rate)
def calculate_payment(litres, fat, snf):
    """Calculate payment for a valid milk entry."""
    return money(litres * calculate_rate(fat, snf))
def validate_entry(litres, fat, snf):
    """Validate the farmer's milk entry."""
    return (litres > 0 and MIN_FAT <= fat <= MAX_FAT and snf > 0)
def main():
    total = Decimal("0.00")
    n = int(input("Enter number of farmers: "))
    for i in range(n):
        print("\\nFarmer", i + 1)
        farmer_id = input("Enter Farmer ID: ")
        litres = Decimal(input("Enter Litres: "))
        fat = Decimal(input("Enter Fat: "))
        snf = Decimal(input("Enter SNF: "))
        if validate_entry(litres, fat, snf):
            rate = calculate_rate(fat, snf)
            amount = calculate_payment(litres, fat, snf)
            total += amount
            print("Farmer ID:", farmer_id)
            print("Rate per litre:", rate)
            print("Payment:", amount)
        else:
            print("Invalid entry. Payment not calculated.")
    print("\\nTen-Day Total:", money(total))
if __name__ == "__main__":
    main()

C. Adapted Python implementation (2 Marks)
from decimal import Decimal, ROUND_HALF_UP
STEP = Decimal("0.1")
FAT_STANDARD = Decimal("3.5")
SNF_STANDARD = Decimal("8.5")
BONUS_RATE = Decimal("0.50")
ORIGINAL_RATE = {
    "base": Decimal("30.00"),
    "fat": Decimal("0.40"),
    "snf": Decimal("0.30")
}
SEASONAL_RATE = {
    "base": Decimal("32.00"),
    "fat": Decimal("0.45"),
    "snf": Decimal("0.30")
}
def money(value):
    """Round money to two decimal places."""
    return value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
def get_rate_card(month):
    """Select the rate card based on the month."""
    if month in (4, 5, 6):
        return SEASONAL_RATE
    return ORIGINAL_RATE
def calculate_rate(fat, snf, month):
    """Calculate the rate using the selected seasonal card."""
    card = get_rate_card(month)
    fat_steps = ((fat - FAT_STANDARD) / STEP).quantize(
        Decimal("1"), rounding=ROUND_HALF_UP)
    snf_steps = ((snf - SNF_STANDARD) / STEP).quantize(
        Decimal("1"), rounding=ROUND_HALF_UP)
    rate = (card["base"] + fat_steps * card["fat"]
            + snf_steps * card["snf"])
    return money(rate)

def calculate_bonus(total_litres):
    """Pay bonus on all litres only when total exceeds 500."""
    if total_litres > Decimal("500"):
        return money(total_litres * BONUS_RATE)

def ten_day_total(entries, month):
    """Return litres, milk payment, bonus and final total."""
    total_litres = sum((e[0] for e in entries), Decimal("0"))
    milk_payment = sum(
        (calculate_payment(l, f, s, month) for l, f, s in entries),
        Decimal("0")
    )
    bonus = calculate_bonus(total_litres)
    return (total_litres, money(milk_payment), bonus,
            money(milk_payment + bonus))
